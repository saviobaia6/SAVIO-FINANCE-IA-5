```ts
export interface FinanceSummary {
  totalBalance: number;
  totalIncomeReceived: number;
  totalIncomeExpected: number;
  totalExpenses: number;
  totalExpensesExpected: number;
  totalOpenInvoices: number;
  totalDebt: number;
  monthlyCommitments: number;
  projectedFreeBalance: number;
}

export interface FinanceConsolidationInput {
  accounts: any[];
  transactions: any[];
  incomes: any[];
  cards: any[];
  debts: any[];
}

export function consolidateFinance(
  data: FinanceConsolidationInput
): FinanceSummary {
  const totalBalance = data.accounts.reduce(
    (sum, account) => sum + (account.balance ?? 0),
    0
  );

  const totalIncomeReceived = data.incomes
    .filter(income => income.status === "received")
    .reduce((sum, income) => sum + (income.amount ?? 0), 0);

  const totalIncomeExpected = data.incomes
    .filter(income => income.status === "expected")
    .reduce((sum, income) => sum + (income.amount ?? 0), 0);

  const totalExpenses = data.transactions
    .filter(transaction =>
      transaction.type === "expense" &&
      transaction.status !== "cancelled" &&
      !transaction.isInternalTransfer &&
      !transaction.isCardBillPayment
    )
    .reduce((sum, transaction) => sum + (transaction.amount ?? 0), 0);

  const totalExpensesExpected = data.transactions
    .filter(transaction =>
      transaction.type === "expense" &&
      transaction.status === "expected"
    )
    .reduce((sum, transaction) => sum + (transaction.amount ?? 0), 0);

  const totalOpenInvoices = data.cards.reduce(
    (sum, card) => sum + (card.openInvoiceAmount ?? 0),
    0
  );

  const totalDebt = data.debts.reduce(
    (sum, debt) => sum + (debt.remainingBalance ?? 0),
    0
  );

  const monthlyCommitments = data.debts.reduce(
    (sum, debt) => sum + (debt.monthlyInstallment ?? 0),
    0
  );

  const projectedFreeBalance =
    totalBalance +
    totalIncomeExpected -
    totalExpensesExpected -
    totalOpenInvoices -
    monthlyCommitments;

  return {
    totalBalance,
    totalIncomeReceived,
    totalIncomeExpected,
    totalExpenses,
    totalExpensesExpected,
    totalOpenInvoices,
    totalDebt,
    monthlyCommitments,
    projectedFreeBalance
  };
}
```

### Atenção técnica

O código acima é uma referência inicial, não deve ser copiado cegamente.

Audite os cálculos antes de integrá-los.

Não subtraia faturas, dívidas ou compromissos duas vezes.

Não misture saldo atual com fluxo mensal.

Não misture receitas previstas com receitas recebidas.

Não utilize `totalBalance` como sinônimo de dinheiro livre.

A projeção deve considerar datas, pagamentos, transferências, faturas, compromissos e o período selecionado.

Se a estrutura atual possuir um `BalanceCalculator`, `consolidationEngine` ou outro motor financeiro, reutilize-o e amplie suas funcionalidades, evitando criar um segundo motor concorrente.

```ts
export interface AccountAnalytics {
  accountId: string;
  currentBalance: number;
  totalIncome: number;
  totalExpenses: number;
  transactionCount: number;
}

export function calculateAccountAnalytics(
  accountId: string,
  transactions: any[],
  accounts: any[]
): AccountAnalytics {
  const account = accounts.find(
    item => item.id === accountId
  );

  const accountTransactions = transactions.filter(
    transaction =>
      transaction.accountId === accountId &&
      transaction.status !== "cancelled"
  );

  const totalIncome = accountTransactions
    .filter(transaction =>
      transaction.type === "income" &&
      !transaction.isInternalTransfer
    )
    .reduce(
      (sum, transaction) => sum + transaction.amount,
      0
    );

  const totalExpenses = accountTransactions
    .filter(transaction =>
      transaction.type === "expense" &&
      !transaction.isInternalTransfer &&
      !transaction.isCardBillPayment
    )
    .reduce(
      (sum, transaction) => sum + transaction.amount,
      0
    );

  return {
    accountId,
    currentBalance: account?.balance ?? 0,
    totalIncome,
    totalExpenses,
    transactionCount: accountTransactions.length
  };
}
```

## Cálculo de limite

```ts
export interface CardLimitAnalysis {
  totalLimit: number;
  usedLimit: number;
  availableLimit: number;
  usagePercentage: number;
}

export function calculateCardLimit(
  totalLimit: number,
  usedLimit: number
): CardLimitAnalysis {
  const safeLimit = Math.max(totalLimit, 0);
  const safeUsed = Math.max(usedLimit, 0);

  const availableLimit = Math.max(
    safeLimit - safeUsed,
    0
  );

  const usagePercentage =
    safeLimit > 0
      ? Math.min((safeUsed / safeLimit) * 100, 100)
      : 0;

  return {
    totalLimit: safeLimit,
    usedLimit: safeUsed,
    availableLimit,
    usagePercentage
  };
}
```

```ts
export interface DebtAnalytics {
  totalDebt: number;
  totalPaid: number;
  totalMonthlyInstallments: number;
  debtCount: number;
}

export function calculateDebtAnalytics(
  debts: any[]
): DebtAnalytics {
  const activeDebts = debts.filter(
    debt => debt.status !== "paid" &&
             debt.status !== "cancelled"
  );

  return {
    totalDebt: activeDebts.reduce(
      (sum, debt) => sum + (debt.remainingBalance ?? 0),
      0
    ),

    totalPaid: debts.reduce(
      (sum, debt) => sum + (debt.totalPaid ?? 0),
      0
    ),

    totalMonthlyInstallments: activeDebts.reduce(
      (sum, debt) => sum + (debt.monthlyInstallment ?? 0),
      0
    ),

    debtCount: activeDebts.length
  };
}
```
